//双均线方向确认器  文华麦语言
//逻辑：交叉发生 且 交叉后双均线同向延伸，才标记"确认"
MA10:MA(CLOSE,10),COLORYELLOW;
MA20:MA(CLOSE,20),COLORWHITE;

//基础交叉
JC:=CROSSUP(MA10,MA20);
SC:=CROSSDOWN(MA10,MA20);

//同向延伸确认（交叉点之后，均线仍在同方向走）
JC_OK:=JC AND MA10>REF(MA10,1) AND MA20>REF(MA20,1);
SC_OK:=SC AND MA10<REF(MA10,1) AND MA20<REF(MA20,1);

DRAWTEXT(JC_OK,LOW-A*0.2,'多确认'),COLORRED;
DRAWTEXT(SC_OK,HIGH+A*0.2,'空确认'),COLORGREEN;

//未确认的交叉画小叉，提示"噪音"
DRAWICON(JC AND NOT(JC_OK),LOW,'ICO1');
DRAWICON(SC AND NOT(SC_OK),HIGH,'ICO2');
